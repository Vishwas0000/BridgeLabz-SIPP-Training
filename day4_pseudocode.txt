CLASS Account:
    METHOD __init__(self, account_number, holder_name, balance):
        SET self.account_number = account_number
        SET self.holder_name = holder_name
        SET self.balance = balance

    METHOD deposit(self, amount):
        IF amount > 0:
            self.balance += amount
            PRINT "Deposited:", amount
        ELSE:
            PRINT "Invalid deposit amount"

    METHOD withdraw(self, amount):
        PRINT "Base withdraw method — to be overridden"

    METHOD display_balance(self):
        PRINT "Account:", self.account_number
        PRINT "Holder:", self.holder_name
        PRINT "Balance:", self.balance


CLASS SavingAccount EXTENDS Account:
    METHOD __init__(self, account_number, holder_name, balance):
        CALL super().__init__(account_number, holder_name, balance)
        SET self.min_balance = 500

    METHOD withdraw(self, amount):
        IF self.balance - amount < self.min_balance:
            PRINT "Cannot withdraw. Minimum balance of", self.min_balance, "required."
        ELSE:
            self.balance -= amount
            PRINT "Withdrawn:", amount


CLASS CurrentAccount EXTENDS Account:
    METHOD __init__(self, account_number, holder_name, balance):
        CALL super().__init__(account_number, holder_name, balance)
        SET self.overdraft_limit = 1000  # Can go negative up to this limit

    METHOD withdraw(self, amount):
        IF self.balance - amount < -self.overdraft_limit:
            PRINT "Cannot withdraw. Overdraft limit exceeded."
        ELSE:
            self.balance -= amount
            PRINT "Withdrawn:", amount


# Example Usage:
CREATE savings = SavingAccount("SA123", "Alice", 1000)
savings.withdraw(600)    # Allowed: balance becomes 400 → should be denied
savings.withdraw(400)    # Allowed: balance becomes 600 → valid

CREATE current = CurrentAccount("CA456", "Bob", 500)
current.withdraw(1300)   # Allowed: overdraft to -800
current.withdraw(300)    # Denied: overdraft limit exceeded
